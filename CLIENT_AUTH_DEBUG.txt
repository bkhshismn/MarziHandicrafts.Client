using System.Net.Http.Headers;
using Marzi.Client.Core.Authentication;

namespace Marzi.Client.Infrastructure.Http;

public sealed class AuthorizationHandler : DelegatingHandler
{
    private readonly ITokenStorage _tokenStorage;

    public AuthorizationHandler(
        ITokenStorage tokenStorage)
    {
        _tokenStorage = tokenStorage;
    }

    protected override async Task<HttpResponseMessage> SendAsync(
        HttpRequestMessage request,
        CancellationToken cancellationToken)
    {
        Console.WriteLine(
            $"[AUTH] {request.Method} {request.RequestUri}");

        string? token = null;

        try
        {
            token =
                await _tokenStorage.GetAccessTokenAsync();
        }
        catch (InvalidOperationException ex)
        {
            Console.WriteLine(
                $"[AUTH] InvalidOperation: {ex.Message}");
        }
        catch (Microsoft.JSInterop.JSException ex)
        {
            Console.WriteLine(
                $"[AUTH] JSException: {ex.Message}");
        }

        Console.WriteLine(
            $"[AUTH] Token exists: {!string.IsNullOrWhiteSpace(token)}");

        if (!string.IsNullOrWhiteSpace(token))
        {
            request.Headers.Authorization =
                new AuthenticationHeaderValue(
                    "Bearer",
                    token);

            Console.WriteLine(
                "[AUTH] Bearer token attached.");
        }
        else
        {
            Console.WriteLine(
                "[AUTH] No access token.");
        }

        return await base.SendAsync(
            request,
            cancellationToken);
    }
}

using Marzi.Client.Core.Authentication;
using Marzi.Client.Infrastructure.Authentication;
using Marzi.Client.Infrastructure.Services;
using Microsoft.JSInterop;
using System.Net;
using System.Net.Http.Headers;

namespace Marzi.Client.Infrastructure.Http;

public sealed class RefreshTokenHandler : DelegatingHandler
{
    private readonly AuthService _authService;
    private readonly ITokenStorage _tokenStorage;
    private readonly TokenRefreshCoordinator _coordinator;

    public RefreshTokenHandler(
        AuthService authService,
        ITokenStorage tokenStorage,
        TokenRefreshCoordinator coordinator)
    {
        _authService = authService;
        _tokenStorage = tokenStorage;
        _coordinator = coordinator;
    }

    protected override async Task<HttpResponseMessage> SendAsync(
        HttpRequestMessage request,
        CancellationToken cancellationToken)
    {
        var response =
            await base.SendAsync(
                request,
                cancellationToken);

        if (response.StatusCode != HttpStatusCode.Unauthorized)
            return response;

        // درخواست‌های authentication نباید وارد refresh شوند.
        if (IsAuthRequest(request))
            return response;

        string? currentToken;

        try
        {
            currentToken =
                await _tokenStorage.GetAccessTokenAsync();
        }
        catch (InvalidOperationException)
        {
            // localStorage/JS هنوز در دسترس نیست.
            return response;
        }
        catch (JSException)
        {
            // JS interop هنوز آماده نیست.
            return response;
        }

        if (string.IsNullOrWhiteSpace(currentToken))
            return response;

        var requestToken =
            request.Headers.Authorization?.Parameter;

        // اگر این درخواست اصلاً با token ارسال نشده،
        // در این مرحله refresh نکن.
        if (string.IsNullOrWhiteSpace(requestToken))
            return response;

        await _coordinator.Lock.WaitAsync(
            cancellationToken);

        try
        {
            // ممکن است درخواست دیگری قبل از ما
            // token را refresh کرده باشد.
            try
            {
                currentToken =
                    await _tokenStorage.GetAccessTokenAsync();
            }
            catch (InvalidOperationException)
            {
                return response;
            }
            catch (JSException)
            {
                return response;
            }

            if (string.IsNullOrWhiteSpace(currentToken))
                return response;

            // اگر token فعلی با token درخواست متفاوت است،
            // یعنی احتمالاً درخواست دیگری قبلاً refresh کرده.
            if (!string.Equals(
                    currentToken,
                    requestToken,
                    StringComparison.Ordinal))
            {
                var retryRequest =
                    await CloneRequestAsync(
                        request,
                        cancellationToken);

                retryRequest.Headers.Authorization =
                    new AuthenticationHeaderValue(
                        "Bearer",
                        currentToken);

                response.Dispose();

                return await base.SendAsync(
                    retryRequest,
                    cancellationToken);
            }

            // فقط وقتی همان token قبلی هنوز فعال است،
            // refresh را انجام بده.
            bool refreshed;

            try
            {
                refreshed =
                    await _authService.RefreshAsync();
            }
            catch (InvalidOperationException)
            {
                return response;
            }
            catch (JSException)
            {
                return response;
            }

            if (!refreshed)
                return response;

            try
            {
                currentToken =
                    await _tokenStorage.GetAccessTokenAsync();
            }
            catch (InvalidOperationException)
            {
                return response;
            }
            catch (JSException)
            {
                return response;
            }

            if (string.IsNullOrWhiteSpace(currentToken))
                return response;

            response.Dispose();

            var retry =
                await CloneRequestAsync(
                    request,
                    cancellationToken);

            retry.Headers.Authorization =
                new AuthenticationHeaderValue(
                    "Bearer",
                    currentToken);

            return await base.SendAsync(
                retry,
                cancellationToken);
        }
        finally
        {
            _coordinator.Lock.Release();
        }
    }

    private static bool IsAuthRequest(
        HttpRequestMessage request)
    {
        var path =
            request.RequestUri?.AbsolutePath
            ?? string.Empty;

        return path.StartsWith(
            "/api/auth/",
            StringComparison.OrdinalIgnoreCase);
    }

    private static async Task<HttpRequestMessage>
        CloneRequestAsync(
            HttpRequestMessage request,
            CancellationToken cancellationToken)
    {
        var clone =
            new HttpRequestMessage(
                request.Method,
                request.RequestUri)
            {
                Version = request.Version,
                VersionPolicy = request.VersionPolicy
            };

        foreach (var header in request.Headers)
        {
            clone.Headers.TryAddWithoutValidation(
                header.Key,
                header.Value);
        }

        if (request.Content is not null)
        {
            var content =
                await request.Content.ReadAsByteArrayAsync(
                    cancellationToken);

            var clonedContent =
                new ByteArrayContent(content);

            foreach (var header in request.Content.Headers)
            {
                clonedContent.Headers.TryAddWithoutValidation(
                    header.Key,
                    header.Value);
            }

            clone.Content = clonedContent;
        }

        return clone;
    }
}

using Marzi.Client.Core.Authentication;
using Marzi.Client.Core.Requests;
using Marzi.Client.Core.Responses;
using Marzi.Client.Infrastructure.Api;
using Refit;

namespace Marzi.Client.Infrastructure.Services;

public sealed class AuthService
{
    private readonly IAuthApi _api;
    private readonly ITokenStorage _tokenStorage;
    private readonly IAuthenticationNotifier _authenticationNotifier;

    public AuthService(
        IAuthApi api,
        ITokenStorage tokenStorage,
        IAuthenticationNotifier authenticationNotifier)
    {
        _api = api;
        _tokenStorage = tokenStorage;
        _authenticationNotifier = authenticationNotifier;
    }

    public async Task<AuthResponse?> LoginAsync(
        LoginRequest request,
        CancellationToken cancellationToken = default)
    {
        try
        {
            var response = await _api.LoginAsync(request);

            if (!response.IsSuccessStatusCode ||
                response.Content is null)
            {
                return null;
            }

            await SaveAuthResponseAsync(response.Content);

            _authenticationNotifier.NotifyAuthenticated(
                response.Content.Token);

            return response.Content;
        }
        catch (ApiException)
        {
            return null;
        }
    }

    public async Task<AuthResponse?> RegisterAsync(
        RegisterRequest request,
        CancellationToken cancellationToken = default)
    {
        try
        {
            var response = await _api.RegisterAsync(request);

            if (!response.IsSuccessStatusCode ||
                response.Content is null)
            {
                return null;
            }

            await SaveAuthResponseAsync(response.Content);

            _authenticationNotifier.NotifyAuthenticated(
                response.Content.Token);

            return response.Content;
        }
        catch (ApiException)
        {
            return null;
        }
    }

    public Task<string?> GetAccessTokenAsync()
    {
        return _tokenStorage.GetAccessTokenAsync();
    }

    public async Task<bool> IsLoggedInAsync()
    {
        var token =
            await _tokenStorage.GetAccessTokenAsync();

        return !string.IsNullOrWhiteSpace(token);
    }

    public async Task<bool> RefreshAsync()
    {
        var refreshToken =
            await _tokenStorage.GetRefreshTokenAsync();

        if (string.IsNullOrWhiteSpace(refreshToken))
        {
            await ClearSessionAsync();
            return false;
        }

        try
        {
            var response = await _api.RefreshAsync(
                new RefreshRequest
                {
                    Token = refreshToken
                });

            if (!response.IsSuccessStatusCode ||
                response.Content is null)
            {
                await ClearSessionAsync();
                return false;
            }

            await SaveAuthResponseAsync(response.Content);

            _authenticationNotifier.NotifyAuthenticated(
                response.Content.Token);

            return true;
        }
        catch (ApiException)
        {
            await ClearSessionAsync();
            return false;
        }
    }

    public async Task LogoutAsync()
    {
        var accessToken =
            await _tokenStorage.GetAccessTokenAsync();

        var refreshToken =
            await _tokenStorage.GetRefreshTokenAsync();

        try
        {
            if (!string.IsNullOrWhiteSpace(accessToken))
            {
                await _api.LogoutAsync(
                    $"Bearer {accessToken}");
            }
            else if (!string.IsNullOrWhiteSpace(refreshToken))
            {
                await _api.RevokeAsync(
                    new RefreshRequest
                    {
                        Token = refreshToken
                    });
            }
        }
        catch (ApiException)
        {
            // حتی اگر Backend در دسترس نباشد،
            // session محلی باید پاک شود.
        }
        finally
        {
            await ClearSessionAsync();
        }
    }
    private Task SaveAuthResponseAsync(
        AuthResponse response)
    {
        return _tokenStorage.SaveTokensAsync(
            response.Token,
            response.RefreshToken);
    }

    private async Task ClearSessionAsync()
    {
        await _tokenStorage.ClearAsync();

        _authenticationNotifier.NotifyLoggedOut();
    }
}

using Marzi.Client.Core.Interfaces;
using Microsoft.JSInterop;

namespace Marzi.Client.Infrastructure.Storage;

public class LocalStorage : ILocalStorage
{
    private readonly IJSRuntime _js;

    public LocalStorage(IJSRuntime js)
    {
        _js = js;
    }

    public async Task SetAsync<T>(string key, T value)
    {
        var json = System.Text.Json.JsonSerializer.Serialize(value);
        await _js.InvokeVoidAsync("localStorage.setItem", key, json);
    }

    public async Task<T?> GetAsync<T>(string key)
    {
        var json = await _js.InvokeAsync<string>("localStorage.getItem", key);
        if (string.IsNullOrEmpty(json))
            return default;

        return System.Text.Json.JsonSerializer.Deserialize<T>(json);
    }

    public async Task RemoveAsync(string key)
    {
        await _js.InvokeVoidAsync("localStorage.removeItem", key);
    }
}

using System.Security.Claims;
using Marzi.Client.Core.Authentication;
using Microsoft.AspNetCore.Components.Authorization;
using Microsoft.JSInterop;

namespace Marzi.Client.Web.Authentication;

public sealed class CustomAuthenticationStateProvider
    : AuthenticationStateProvider,
      IAuthenticationNotifier
{
    private readonly ITokenStorage _tokenStorage;
    private readonly IAuthenticationTokenStore _tokenStore;

    private static readonly ClaimsPrincipal Anonymous =
        new(new ClaimsIdentity());

    private bool _initialized;

    public CustomAuthenticationStateProvider(
        ITokenStorage tokenStorage,
        IAuthenticationTokenStore tokenStore)
    {
        _tokenStorage = tokenStorage;
        _tokenStore = tokenStore;
    }

    public override async Task<AuthenticationState>
        GetAuthenticationStateAsync()
    {
        try
        {
            var accessToken =
                await _tokenStorage.GetAccessTokenAsync();

            if (string.IsNullOrWhiteSpace(accessToken))
                return CreateAnonymousState();

            return CreateAuthenticatedState(accessToken);
        }
        catch (InvalidOperationException)
        {
            return CreateAnonymousState();
        }
        catch (JSException)
        {
            return CreateAnonymousState();
        }
        catch
        {
            return CreateAnonymousState();
        }
    }

    public async Task InitializeAsync()
    {
        if (_initialized)
            return;

        try
        {
            await _tokenStorage.RestoreAsync();

            var accessToken =
                await _tokenStorage.GetAccessTokenAsync();

            var refreshToken =
                await _tokenStorage.GetRefreshTokenAsync();

            if (!string.IsNullOrWhiteSpace(accessToken))
            {
                if (!string.IsNullOrWhiteSpace(refreshToken))
                {
                    _tokenStore.SetTokens(
                        accessToken,
                        refreshToken);
                }
                else
                {
                    _tokenStore.SetAccessToken(
                        accessToken);
                }
            }

            _initialized = true;

            var state =
                await GetAuthenticationStateAsync();

            Console.WriteLine(
                $"[AUTH INIT] IsAuthenticated: {state.User.Identity?.IsAuthenticated}");

            Console.WriteLine(
                $"[AUTH INIT] IsAdmin: {state.User.IsInRole("Admin")}");

            NotifyAuthenticationStateChanged(
                Task.FromResult(state));
        }
        catch (InvalidOperationException)
        {
            // JS هنوز آماده نیست.
            // در تلاش بعدی دوباره Restore می‌شود.
            return;
        }
        catch (JSException)
        {
            return;
        }
    }

    public void NotifyAuthenticated(
        string accessToken)
    {
        _tokenStore.SetAccessToken(accessToken);

        var state =
            CreateAuthenticatedState(accessToken);

        NotifyAuthenticationStateChanged(
            Task.FromResult(state));
    }

    public void NotifyLoggedOut()
    {
        _initialized = true;

        _tokenStore.Clear();

        NotifyAuthenticationStateChanged(
            Task.FromResult(
                CreateAnonymousState()));
    }

    public async Task RefreshAsync()
    {
        var state =
            await GetAuthenticationStateAsync();

        NotifyAuthenticationStateChanged(
            Task.FromResult(state));
    }

    private static AuthenticationState
        CreateAuthenticatedState(string accessToken)
    {
        var claims =
            JwtClaimsParser.ParseClaimsFromJwt(accessToken)
                .ToList();

        Console.WriteLine(
            $"[AUTH STATE] Claims count: {claims.Count}");

        foreach (var claim in claims)
        {
            Console.WriteLine(
                $"[AUTH STATE] {claim.Type} = {claim.Value}");
        }

        var identity =
            new ClaimsIdentity(
                claims,
                authenticationType: "jwt");

        var principal =
            new ClaimsPrincipal(identity);

        Console.WriteLine(
            $"[AUTH STATE] IsAuthenticated: {principal.Identity?.IsAuthenticated}");

        Console.WriteLine(
            $"[AUTH STATE] IsAdmin: {principal.IsInRole("Admin")}");

        Console.WriteLine(
            $"[AUTH STATE] IsInstructor: {principal.IsInRole("Instructor")}");

        return new AuthenticationState(principal);
    }

    private static AuthenticationState
        CreateAnonymousState()
    {
        return new AuthenticationState(
            Anonymous);
    }
}
